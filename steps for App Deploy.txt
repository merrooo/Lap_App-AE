Step 1: Create Project Folder & Navigate
# Create folder (if not already created)
mkdir power-factor-app

# Navigate into it
cd power-factor-app
------------------
Step 2: Initialize package.json
npm init -y
Optional: Edit package.json to customize:
{
  "name": "power-factor-app",
  "version": "1.0.0",
  "description": "Power Factor Calculator App",
  "main": "index.js",
  "scripts": {
    "build": "echo 'Build complete'"
  },
  "author": "Your Name",
  "license": "MIT"
}
------------------
Step 3: Create capacitor.config.json
{
  "appId": "com.powerfactor.app",
  "appName": "Power Factor App",
  "webDir": "www",
  "bundledWebRuntime": false,
  "server": {
    "androidScheme": "https"
  }
}
------------------
Step 4: Create ionic.config.json
{
  "name": "Power Factor App",
  "type": "custom",
  "integrations": {
    "capacitor": {}
  }
}
------------------
Step 5: Create App Folder & Required Files
# Create for Example (PF folder)
mkdir PF
------------------
Create index.html
------------------
Create manifest.json
------------------
Create sw.js
------------------
Step 6: Install Dependencies
# Install Ionic CLI globally (if not installed)
npm install -g @ionic/cli

# Install Capacitor locally
npm install @capacitor/core @capacitor/cli @capacitor/android

# Install Capacitor assets plugin
npm install @capacitor/assets
------------------
Step 7: Initialize Ionic Project (if not already)
ionic init --type=custom
------------------
Step 8: Add Android Platform
npx cap add android
------------------
Step 9: Copy Web Assets to Android
npx cap copy
------------------
Step 10 Sync Your Project
npx cap sync android
------------------
Step 11: Build APK
cd android
.\gradlew.bat assembleDebug
cd ..

# Create all files

{

  "version": "1.0.0",
  "description": "Power Factor Calculator App",
  "main": "index.js",
  "scripts": {
    "build": "echo 'Build complete'"
  },
  "author": "",
  "license": "MIT"
}
EOF

cat > capacitor.config.json << 'EOF'
{
  "appId": "com.powerfactor.app",
  "appName": "Power Factor App",
  "webDir": "www",
  "bundledWebRuntime": false
}
EOF

cat > ionic.config.json << 'EOF'
{
  "name": "Power Factor App",
  "type": "custom",
  "integrations": {
    "capacitor": {}
  }
}
EOF

mkdir -p www
cd www

cat > index.html << 'EOF'
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Power Factor App</title>
  <link rel="manifest" href="manifest.json">
</head>
<body>
  <h1>⚡ Power Factor App</h1>
  <p>Your app is ready!</p>
</body>
</html>
EOF

cat > manifest.json << 'EOF'
{
  "name": "Power Factor App",
  "short_name": "PowerFactor",
  "start_url": "index.html",
  "display": "standalone",
  "background_color": "#ffffff",
  "theme_color": "#4CAF50"
}
EOF

touch sw.js

cd ..
