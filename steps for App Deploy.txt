Step 1: Create Project Folder & Navigate
# Create folder (if not already created)
mkdir power-factor-app

# Navigate into it
cd power-factor-app
------------------
Step 2: Initialize package.json
npm init -y
Optional: Edit package.json to customize:
{
  "name": "power-factor-app",
  "version": "1.0.0",
  "description": "Power Factor Calculator App",
  "main": "index.html",
 "scripts": {
    "build": "echo 'No web compilation needed'",
    "ionic:build": "npm run build",
    "fix:cap": "node ./scripts/fix-cap-bin.js",
    "cap": "node ./node_modules/@capacitor/cli/bin/capacitor",
    "sync": "node ./node_modules/@capacitor/cli/bin/capacitor sync",
    "sync:android": "node ./node_modules/@capacitor/cli/bin/capacitor sync android",
    "postinstall": "node ./scripts/fix-cap-bin.js && node ./node_modules/@capacitor/cli/bin/capacitor sync android"
  },
  "dependencies": {
    "@capacitor/android": "^8.4.2",
    "@capacitor/cli": "^8.4.2",
    "@capacitor/core": "^8.4.2"
  }
}
------------------
Step 3: Create capacitor.config.json
{
  "appId": "com.powerfactor.app",
  "appName": "Power Factor App",
  "webDir": "PF",
  "bundledWebRuntime": false,
    "npmClient": "npm",
   "plugins": {
    "SplashScreen": {
      "launchShowDuration": 0
    }
  },
  "android": {
    "allowMixedContent": true
  }
}
------------------
Step 4: Create ionic.config.json
{
  "name": "Power Factor App",
  "type": "custom",
  "integrations": {
    "capacitor": {}
  }
}
------------------
Step 5: Create App Folder & Required Files
# Create for Example (PF folder)
mkdir PF
------------------
Create index.html
------------------
Create manifest.json
------------------
Create sw.js
------------------
Step 6: Install Dependencies
# Install Ionic CLI globally (if not installed)
npm install -g @ionic/cli

# Install Capacitor locally(node_modules)
npm install 

# Install Capacitor assets plugin
npm install @capacitor/assets
------------------
Step 7: Initialize Ionic Project (if not already)
ionic init --type=custom
------------------
Step 8: Add Android Platform(android)
npx cap add android
------------------
Step 9: Copy Web Assets to Android
npx cap copy
------------------
Step 10 Sync Your Project
npx cap sync android
------------------
